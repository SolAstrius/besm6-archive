 672  Q(IQP1+I)=Q(IQP1+I)*BETA                                             00288
      DO 500 I=1,N                                                         00289
  500 Q(IQP1+I)=Q(IQP1+I)+Q(IQR+I)                                         00290

    7 CONTINUE                                                             00292
